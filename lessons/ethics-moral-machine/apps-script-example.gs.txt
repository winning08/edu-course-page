const FIELD_KEYS = [
  "result_saved",
  "result_sacrificed",
  "preference_count",
  "preference_passenger",
  "preference_law",
  "preference_inaction",
  "preference_gender",

function doPost(e) {
  const lock = LockService.getScriptLock();

  try {
    const spreadsheet = SpreadsheetApp.openById(SPREADSHEET_ID);
    let sheet = spreadsheet.getSheetByName(SHEET_NAME);
    if (!sheet) sheet = spreadsheet.insertSheet(SHEET_NAME);
    if (sheet.getLastRow() === 0) sheet.appendRow(HEADERS);

    const data = JSON.parse(e.postData.contents);
    const studentNumber = String(data.student_number || "").trim();
    const studentName = String(data.student_name || "").trim();
    const answers = FIELD_KEYS.map((key) => String(data[key] || "").trim());

    if (
      !/^\d{5}$/.test(studentNumber) ||
      !studentName ||
      answers.some((answer) => !answer)
    ) {
      return jsonResponse(false, "필수 응답을 모두 입력해 주세요.");
    }

    lock.waitLock(10000);
    sheet.appendRow([
      new Date(),
      safeCell(studentNumber),
      safeCell(studentName),
      ...answers.map(safeCell),
    ]);

    return jsonResponse(true, "제출이 완료되었습니다.");
  } catch (error) {
    return jsonResponse(false, "제출 중 오류가 발생했습니다.");
  } finally {
    if (lock.hasLock()) lock.releaseLock();
  }
}

function safeCell(value) {
  const text = String(value || "");
  return /^[=+\-@]/.test(text) ? "'" + text : text;
}

function jsonResponse(success, message) {
  return ContentService.createTextOutput(
    JSON.stringify({ success: success, message: message }),
  ).setMimeType(ContentService.MimeType.JSON);
}

// Google Apps Script 예시 코드 (teacher setup용).
// 사용법: 스프레드시트(https://docs.google.com/spreadsheets/d/1Qt4IjBuwMr2PgacQ1cyyeXjJjwJxph1YjBJ9ffskQHE/edit)의
// 확장 프로그램 → Apps Script에서 이 내용을 그대로 붙여넣고, "제출"이라는 이름의
// 시트를 먼저 만들어 둔 뒤 웹 앱으로 배포합니다.
// 배포 시 실행 계정은 "나", 액세스 권한은 "모든 사용자"로 설정해야 학생 브라우저에서
// 로그인 없이 제출할 수 있습니다.
//
// 이 파일은 .gs가 아니라 .gs.txt로 저장되어 있습니다 — 이 저장소는 정적 사이트
// 빌드 대상이 아니라 "복사해서 Apps Script 편집기에 붙여넣는 참고 코드"이기 때문입니다.
// "제출" 시트의 첫 행에는 다음 열 이름을 순서대로 만들어 두세요.
// 제출시각 | 학번 | 이름 | 질문1 | 이유1 | 추가 질문 | 추가 질문 이유

function doPost(e) {
  const lock = LockService.getScriptLock();

  try {
    const sheet = SpreadsheetApp
      .getActiveSpreadsheet()
      .getSheetByName("제출");

    if (!sheet) {
      throw new Error('"제출" 시트를 찾을 수 없습니다.');
    }

    const data = JSON.parse(e.postData.contents);

    const studentNumber = String(data.student_number || "").trim();
    const studentName = String(data.student_name || "").trim();
    const question = String(data.question || "").trim();
    const reason = String(data.reason || "").trim();
    const followup = String(data.followup || "").trim();
    const followupReason = String(data.followup_reason || "").trim();

    if (!studentNumber || !studentName || !question || !reason) {
      return ContentService
        .createTextOutput(JSON.stringify({
          success: false,
          message: "학번, 이름, 첫 번째 질문과 이유를 모두 입력하세요."
        }))
        .setMimeType(ContentService.MimeType.JSON);
    }

    if (followup && !followupReason) {
      return ContentService
        .createTextOutput(JSON.stringify({
          success: false,
          message: "추가 질문을 적었다면 그 이유도 함께 적어주세요."
        }))
        .setMimeType(ContentService.MimeType.JSON);
    }

    // 여러 학생이 동시에 제출해도 행 추가가 겹치지 않도록 잠금을 사용합니다.
    lock.waitLock(10000);

    sheet.appendRow([
      new Date(),
      safeCell(studentNumber),
      safeCell(studentName),
      safeCell(question),
      safeCell(reason),
      safeCell(followup),
      safeCell(followupReason)
    ]);

    return ContentService
      .createTextOutput(JSON.stringify({
        success: true,
        message: "제출이 완료되었습니다."
      }))
      .setMimeType(ContentService.MimeType.JSON);

  } catch (error) {
    return ContentService
      .createTextOutput(JSON.stringify({
        success: false,
        message: "제출 중 오류가 발생했습니다."
      }))
      .setMimeType(ContentService.MimeType.JSON);
  } finally {
    if (lock.hasLock()) lock.releaseLock();
  }
}

// =, +, -, @로 시작하는 학생 입력이 스프레드시트 수식으로 실행되지 않게 한다.
function safeCell(value) {
  const text = String(value || "");
  return /^[=+\-@]/.test(text) ? "'" + text : text;
}
